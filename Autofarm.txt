local isServerHopping = false

local function ServerHop()
    if isServerHopping then return end
    isServerHopping = true

    if UpdateUIState then
        UpdateUIState("Hopping (No ATM / Alt)")
    end

    print("[Server Hop] Looking for server (Players: 2 - 20)...")

    local PlaceId = game.PlaceId
    local JobId = game.JobId

    task.spawn(function()
        pcall(function()
            local cursor = ""
            local validServers = {}

            -- Fetch up to 100 public servers
            local req = game:HttpGet("https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/0?sortOrder=Desc&limit=100")
            local body = HttpService:JSONDecode(req)

            if body and body.data then
                for _, server in ipairs(body.data) do
                    if type(server) == "table" and server.id ~= JobId then
                        local playerCount = server.playing or 0
                        local maxPlayers = server.maxPlayers or 40

                        -- Target servers between 2 and 20 players that aren't full
                        if playerCount >= 2 and playerCount <= 25 and playerCount < maxPlayers then
                            table.insert(validServers, server.id)
                        end
                    end
                end
            end

            if #validServers > 0 then
                local targetServer = validServers[math.random(1, #validServers)]
                print("[Server Hop] Teleporting to valid server:", targetServer)
                TeleportService:TeleportToPlaceInstance(PlaceId, targetServer, LocalPlayer)
            else
                print("[Server Hop] No server matching 2-20 players found. Fallback teleport...")
                TeleportService:Teleport(PlaceId, LocalPlayer)
            end
        end)

        task.delay(10, function()
            isServerHopping = false
        end)
    end)
end

local function isAltAccount(player)
    if not player or player == LocalPlayer or not getgenv().AltAccounts then
        return false
    end

    for _, identifier in ipairs(getgenv().AltAccounts) do
        if type(identifier) == "number" and player.UserId == identifier then
            return true
        elseif type(identifier) == "string" and player.Name:lower() == identifier:lower() then
            return true
        end
    end

    return false
end

local function checkAltCollision(player)
    if isAltAccount(player) then
        print("[Alt Detector] Alt found in server: " .. player.Name)

        -- TIE-BREAKER: Higher UserId hops, Lower UserId stays
        if LocalPlayer.UserId > player.UserId then
            print("[Alt Detector] LocalPlayer has higher UserId. Leaving server...")
            task.wait(math.random(1, 2))
            ServerHop()
        else
            print("[Alt Detector] LocalPlayer has lower UserId. Remaining in server.")
        end
    end
end

for _, player in ipairs(Players:GetPlayers()) do
    checkAltCollision(player)
end

Players.PlayerAdded:Connect(function(player)
    checkAltCollision(player)
end)

local rejoinInProgress = false
local teleportAttempted = false

local function initiateRejoinSequence()
    if rejoinInProgress then return end
    rejoinInProgress = true

    task.spawn(function()
        task.wait(2)
        if teleportAttempted then return end
        teleportAttempted = true

        local success, err = pcall(function()
            TeleportService:Teleport(game.PlaceId, LocalPlayer)
        end)

        if not success then
            teleportAttempted = false
            task.wait(5)
            if not rejoinInProgress then return end

            local retrySuccess = pcall(function()
                TeleportService:Teleport(game.PlaceId, LocalPlayer)
            end)

            if not retrySuccess then
                rejoinInProgress = false
                teleportAttempted = false
            end
        end
    end)
end

pcall(function()
    TeleportService.TeleportInitFailed:Connect(function()
        if rejoinInProgress then
            task.spawn(function()
                task.wait(5)
                rejoinInProgress = false
                teleportAttempted = false
                initiateRejoinSequence()
            end)
        end
    end)
end)

pcall(function()
    GuiService.ErrorMessageChanged:Connect(function()
        initiateRejoinSequence()
    end)
end)

task.spawn(function()

    local success = pcall(function()

    end)
    if not success or not promptOverlay then return end

    pcall(function()

            initiateRejoinSequence()
        end
    end)

    pcall(function()
        promptOverlay.ChildAdded:Connect(function(child)
            if child and child.Name == "ErrorPrompt" then
                initiateRejoinSequence()
            end
        end)
    end)
end)

local function getCurrentMoney()
    local leaderstats = LocalPlayer:FindFirstChild("leaderstats")
    local money = leaderstats and leaderstats:FindFirstChild("Money")
    if money then return money.Value end

    local dataFolder = LocalPlayer:FindFirstChild("DataFolder")
    local currency = dataFolder and dataFolder:FindFirstChild("Currency")
    if currency then return currency.Value end

    return 0
end

local function getAmmoCount()
    local dataFolder = LocalPlayer:FindFirstChild("DataFolder")
    local inventory = dataFolder and dataFolder:FindFirstChild("Inventory")
    if inventory then
        for _, v in ipairs(inventory:GetChildren()) do
            if string.find(v.Name, "Drum%-Shotgun") then
                return tonumber(v.Value) or 0
            end
        end
    end
    return 0
end

local function formatNumber(value)
    local left, num, right = string.match(tostring(value), '^([^%d]*)(%d+)(.-)$')
    if not num then return tostring(value) end
    return left .. (num:reverse():gsub('(%d%d%d)', '%1,'):reverse():gsub('^,', '')) .. right
end

local StartTime = os.time()
local StartCash = getCurrentMoney()
local LastWebhookTime = 0

local http_request_func = request or http_request or (syn and syn.request) or (fluxus and fluxus.request)

local function SendWebhookUpdate()
    local currentCash = getCurrentMoney()
    local cashEarned = currentCash - StartCash
    local timeElapsed = os.time() - StartTime
    local hoursElapsed = timeElapsed / 3600
    local cashPerHour = hoursElapsed > 0 and math.floor(cashEarned / hoursElapsed) or 0

    if UI.CashRate then
        UI.CashRate.Text = "$" .. formatNumber(cashPerHour) .. " / hr"
    end

    if not getgenv().WEBHOOK_URL or getgenv().WEBHOOK_URL == "" or not http_request_func then
        return
    end

    if os.time() - LastWebhookTime < (getgenv().WEBHOOK_COOLDOWN or 60) then
        return
    end

    LastWebhookTime = os.time()

    local embedData = {
        ["embeds"] = {{
            ["title"] = "Skido Autofarm Progress Update",
            ["color"] = 65280,
            ["fields"] = {
                { ["name"] = "Account", ["value"] = "`" .. LocalPlayer.Name .. "`", ["inline"] = true },
                { ["name"] = "Total Wallet Balance", ["value"] = "$" .. formatNumber(currentCash), ["inline"] = true },
                { ["name"] = "Cash Earned This Session", ["value"] = "$" .. formatNumber(cashEarned), ["inline"] = false },
                { ["name"] = "Estimated Cash / Hour", ["value"] = "$" .. formatNumber(cashPerHour) .. "/hr", ["inline"] = true },
                { ["name"] = "Time Active", ["value"] = math.floor(timeElapsed / 60) .. " minutes", ["inline"] = true }
            },
            ["footer"] = { ["text"] = "Skido AutoFarm System • " .. os.date("%X") }
        }}
    }

    pcall(function()
        http_request_func({
            Url = getgenv().WEBHOOK_URL,
            Method = "POST",
            Headers = { ["content-type"] = "application/json" },
            Body = HttpService:JSONEncode(embedData)
        })
    end)
end

task.spawn(function()
    task.wait(2)
    SendWebhookUpdate()
    while task.wait(5) do
        SendWebhookUpdate()
    end
end)

local function GetCash()
    local Found = {}
    local ignored = workspace:FindFirstChild("Ignored")
    local dropParent = ignored and ignored:FindFirstChild("Drop")
    if not dropParent then return Found end

    local character = LocalPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    if not hrp then return Found end

    local children = dropParent:GetChildren()
    for i = 1, #children do
        local v = children[i]
        if v.Name == "MoneyDrop" and not pickedUpCash[v] and v:IsA("BasePart") then
            if (v.Position - hrp.Position).Magnitude <= (getgenv().CASH_AURA_RANGE + 20) then
                table.insert(Found, v)
            end
        end
    end
    return Found
end

local shopFolder = workspace:FindFirstChild("Ignored") and workspace.Ignored:FindFirstChild("Shop")

local function findShopItem(nameKeyword)
    if not shopFolder then return nil end
    for _, item in ipairs(shopFolder:GetChildren()) do

local function getPartCFrame(instance)
    if instance:IsA("BasePart") then return instance.CFrame end
    if instance:IsA("Model") then
        if instance.PrimaryPart then return instance.PrimaryPart.CFrame end
        local bp = instance:FindFirstChildWhichIsA("BasePart")
        if bp then return bp.CFrame end
    end
    return nil
end

local function executeLootSequence(hum, hrp, currentLootTarget)
    UpdateUIState("Picking up cash")
    LocalPlayer.CameraMode = Enum.CameraMode.Classic
    hum:UnequipTools()
    task.wait(0.3)

    local pickupStart = tick()
    while (tick() - pickupStart) < 20 do
        if isCuffed or not getgenv().isLooting or (currentLootTarget and getgenv().selectedCashier ~= currentLootTarget) then
            break
        end

        local CashItems = GetCash()
        if #CashItems == 0 then break end

        for i = 1, #CashItems do
            if isCuffed or not getgenv().isLooting or (currentLootTarget and getgenv().selectedCashier ~= currentLootTarget) then
                break
            end

            local v = CashItems[i]
            if v and v.Parent and not pickedUpCash[v] then
                local cd = v:FindFirstChildOfClass("ClickDetector")
                if cd and hrp then
                    pickedUpCash[v] = true
                    while v and v.Parent and _fireclickdetector and not isCuffed do
                        hrp.CFrame = CFrame.new(v.Position - Vector3.new(0, 3, 0))
                        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                        hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)

                        local randomOffset = Vector3.new(math.random(-5, 5), math.random(3, 6), math.random(-5, 5))
                        workspace.CurrentCamera.CFrame = CFrame.lookAt(v.Position + randomOffset, v.Position)

                        _fireclickdetector(cd, 0)
                        task.wait(0.05)
                        if not v or not v.Parent then break end
                    end
                    task.wait(PICKUP_DELAY)
                end
            end
        end
        RunService.RenderStepped:Wait()
    end
    task.wait(0.3)
end

local function IsCashierAvailable(v)
    if not v or not v.Name:match("CA%$HIER") or v.Name:match("VAULT") then return false end
    local hum = v:FindFirstChildOfClass("Humanoid")
    local head = v:FindFirstChild("Head")
    if not hum or not head or hum.Health <= 25 then return false end
    if (head.Position - Vector3.new(-623.998474, 24.9000168, -285.062225)).Magnitude <= 8 then return false end
    return true
end

local function FindAvailableATM()
    local cashiersFolder = workspace:FindFirstChild("Cashiers")
    if not cashiersFolder then return nil end
    for _, v in ipairs(cashiersFolder:GetChildren()) do
        if IsCashierAvailable(v) then return v end
    end
    return nil
end

                -- Christmas Sock Reset
                if char:FindFirstChild("Christmas_Sock") then
                    UpdateUIState("Resetting Character...")
                    LocalPlayer.CameraMode = Enum.CameraMode.Classic
                    getgenv().selectedCashier = nil
                    getgenv().isLooting = false
                    hum.Health = 0
                    task.wait(1)
                    return
                end

                -- Death / K.O Cleanup
                local bodyEffects = char:FindFirstChild("BodyEffects")
                local koObject = bodyEffects and bodyEffects:FindFirstChild("K.O")
                local isKO = koObject and koObject.Value == true

                if hum.Health <= 15 or isKO then
                    UpdateUIState("Dead")
                    LocalPlayer.CameraMode = Enum.CameraMode.Classic
                    getgenv().selectedCashier = nil
                    getgenv().isLooting = false

                    local partsToDestroy = {
                        "Head", "RightHand", "LeftHand", "LowerTorso", "UpperTorso",
                        "LeftLowerLeg", "RightLowerLeg", "RightFoot", "LeftFoot",
                        "LeftUpperLeg", "RightUpperLeg", "RightUpperArm", "RightLowerArm",
                        "LeftLowerArm", "LeftUpperArm"
                    }

                    for _, partName in ipairs(partsToDestroy) do
                        local part = char:FindFirstChild(partName)
                        if part then part:Destroy() end
                    end

                    task.wait(1)
                    return
                end

                -- Gear Requirements
                local currentMoney = getCurrentMoney()
                local canAffordGunfarm = currentMoney >= 20000
                local hasGun = LocalPlayer.Backpack:FindFirstChild("[Drum-Shotgun]") or char:FindFirstChild("[Drum-Shotgun]")
                local outOfAmmo = hasGun and getAmmoCount() <= 0

                if getgenv().Gunfarm and canAffordGunfarm then
                    if not hasGun or getgenv().ammoBoughtCount < 4 or outOfAmmo then
                        getgenv().selectedCashier = nil
                        getgenv().isLooting = false
                        getgenv().noclipCashier = false
                        LocalPlayer.CameraMode = Enum.CameraMode.Classic
                        BuyGunGears()
                        return
                    end
                else
                    if currentMoney < 20000 then
                        getgenv().ammoBoughtCount = 0
                    end
                end

                -- Cashier Target Processing
                local selectedCashier = getgenv().selectedCashier

                if selectedCashier and (selectedCashier:FindFirstChild("Head") or getgenv().isLooting) then
                    
                    if getgenv().isLooting or (selectedCashier:FindFirstChild("Humanoid") and selectedCashier.Humanoid.Health < 1) then
                        if not getgenv().isProcessingCash then
                            getgenv().isProcessingCash = true
                            getgenv().isLooting = true

                            local currentLootTarget = selectedCashier
                            executeLootSequence(hum, hrp, currentLootTarget)

                            if getgenv().selectedCashier == currentLootTarget then
                                getgenv().selectedCashier = nil
                                getgenv().noclipCashier = false
                            end

                            getgenv().isProcessingCash = false
                            getgenv().isLooting = false
                            task.wait(1.3)
                        end
                    elseif selectedCashier:FindFirstChild("Humanoid") and selectedCashier.Humanoid.Health >= 1 then
                        if selectedCashier:FindFirstChild("Open") then
                            local cashierCF = selectedCashier.Open.CFrame
                            local targetPos

                            if getgenv().Gunfarm and canAffordGunfarm and not outOfAmmo then
                                UpdateUIState("Attacking with Gun...")
                                targetPos = (cashierCF * CFrame.new(0, 0.5, 3.4)).Position
                                hrp.CFrame = CFrame.lookAt(targetPos, cashierCF.Position)
                            else
                                UpdateUIState("Attacking with Combat...")
                                targetPos = (cashierCF * CFrame.new(0, 5.5, 0)).Position
                                hrp.CFrame = CFrame.new(targetPos) * CFrame.Angles(math.rad(-90), 0, 0)
                            end

                            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                            hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                        end

                        if not getgenv().noclipCashier then
                            getgenv().noclipCashier = true
                            for _, v in pairs(selectedCashier:GetDescendants()) do
                                if v:IsA("BasePart") then v.CanCollide = false end
                            end
                        end

                        if getgenv().Gunfarm and canAffordGunfarm and not outOfAmmo then
                            LocalPlayer.CameraMode = Enum.CameraMode.LockFirstPerson
                            if selectedCashier:FindFirstChild("Head") then
                                local targetPosition = selectedCashier.Head.Position - Vector3.new(0, 1.6, 0)
                                Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, targetPosition)
                            end
                        else
                            LocalPlayer.CameraMode = Enum.CameraMode.Classic
                        end

                        local weaponName = (getgenv().Gunfarm and canAffordGunfarm and not outOfAmmo) and "[Drum-Shotgun]" or "Combat"
                        local attackTool = LocalPlayer.Backpack:FindFirstChild(weaponName) or char:FindFirstChild(weaponName)

                        if attackTool then
                            if attackTool.Parent == LocalPlayer.Backpack then
                                attackTool.Parent = char
                            end

                            if attackTool:IsA("Tool") and attackTool.Name ~= "Combat" then
                                local clips = attackTool:FindFirstChild("Clips") or attackTool:FindFirstChild("Ammo")
                                if clips and clips.Value == 0 then
                                    game:GetService("ReplicatedStorage").MainEvent:FireServer("Reload", attackTool)
                                    task.wait(0.2)
                                end
                            end

                            local attacking = char:FindFirstChild("BodyEffects") and char.BodyEffects:FindFirstChild("Attacking")
                            if attacking and attacking.Value == false then
                                attackTool:Activate()
                            elseif not char:FindFirstChild("BodyEffects") then
                                attackTool:Activate()
                            end
                        end
                    end
                elseif not getgenv().isLooting then
                    UpdateUIState("Scanning ATM")
                    LocalPlayer.CameraMode = Enum.CameraMode.Classic

                    local foundATM = FindAvailableATM()

                    if foundATM then
                        getgenv().selectedCashier = foundATM
                        UpdateUIState("ATM FOUND")
                    else
                        -- NO ATM FOUND: AUTOMATIC SERVER HOP
                        UpdateUIState("NO ATM - SERVER HOPPING")
                        ServerHop()
                        task.wait(3)
                    end
                end
            end
        end)
    end
end)

local function setupListener(char)
    local be = char:WaitForChild("BodyEffects", 5)
    local cuff = be and be:WaitForChild("Cuff", 5)

        cuff.Changed:Connect(function(val)
            isCuffed = val
            if isCuffed then
                LocalPlayer.CameraMode = Enum.CameraMode.Classic
                getgenv().selectedCashier = nil
                getgenv().isLooting = false
                getgenv().noclipCashier = false

                task.spawn(function()
                    while isCuffed do
                        pcall(BuyAndUseKey)
                        task.wait(0.5)
                    end
                end)
            end
        end)
    end
end

local ModTable = loadstring(game:HttpGet("https://raw.githubusercontent.com/XK5NG/XK5NG.github.io/refs/heads/main/Contents"))()

do
    local Check = function(Player)
        if not Player or Player == LocalPlayer or not Player:IsA("Player") then return end

        local Detect_Type = "Kick"
        local Flags = {}
        local ShouldKick = true

        if ModTable and ModTable[tostring(Player.UserId)] then
            local Reason = "matched"
            local Message = `a mod has joined your game | triggered by: {Player.DisplayName} (@{Player.Name}) | id: {Player.UserId} | reason: {Reason}`

            if Detect_Type == "Notify" and Library then Library:Notify(Message, 15) end
            if Detect_Type == "Kick" then LocalPlayer:Kick(Message) end
            return
        end

        local Success, InGroup = pcall(Player.IsInGroup, Player, 10604500)
        if Success and InGroup then
            local Xp, Role = pcall(Player.GetRoleInGroup, Player, 10604500)
            if Xp and Role == "Roblox Verified" then
                if Detect_Type == "Notify" and Library then
                    Library:Notify(`verified user {Player.DisplayName} (@{Player.Name}) - roblox verified`, 15)
                end
                ShouldKick = false
            end
        end

        for GroupId, Data in pairs({
            [10604500] = { name = "da hood verified", notifies = {"celebs", "roblox verified", "trades"} },
            [17215700] = "stars staff",
            [8068202] = "da hood stars",
            [4698921] = { name = "da hood entertainment", forbidden = {"owner", "admin", "devs", "monetization", "contributed", "moderators"} }
        }) do
            local SuccessGroup, InGroupRole = pcall(Player.IsInGroup, Player, GroupId)
            if SuccessGroup and InGroupRole then
                local Xp, Role = pcall(Player.GetRoleInGroup, Player, GroupId)
                if Xp and Role then
                    if type(Data) == "table" then
                        if Data.notifies and table.find(Data.notifies, string.lower(Role)) then
                            if Detect_Type == "Notify" and Library then
                                Library:Notify(`possible mod detected {Player.DisplayName} (@{Player.Name}) - {Role} in {Data.name}`, 15)
                            end
                        end
                        if Data.forbidden and table.find(Data.forbidden, string.lower(Role)) then
                            table.insert(Flags, `{Data.name} - {Role}`)
                        end
                    else
                        table.insert(Flags, `{Data} - {Role}`)
                    end
                end
            end
        end

        if #Flags > 0 and ShouldKick then
            local Reason = table.concat(Flags, ", ")
            local Message = `a mod was on your game | triggered by: {Player.DisplayName} (@{Player.Name}) | id: {Player.UserId} | reason: {Reason}`
            if Detect_Type == "Notify" and Library then Library:Notify(Message, 15) end
            if Detect_Type == "Kick" then LocalPlayer:Kick(Message) end
        end
    end

    if Player_Data then
        for Player, Data in pairs(Player_Data) do
            Check(Data.Player)
        end
    else
        for _, Player in pairs(Players:GetPlayers()) do
            Check(Player)
        end
    end

    Players.PlayerAdded:Connect(function(Player)
        Check(Player)
    end)
end
