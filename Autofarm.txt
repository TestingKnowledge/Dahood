local function getCurrentMoney()

    local leaderstats = LocalPlayer:FindFirstChild("leaderstats")
    local money = leaderstats and leaderstats:FindFirstChild("Money")

    if money then
        return money.Value
    end

    local dataFolder = LocalPlayer:FindFirstChild("DataFolder")
    local currency = dataFolder and dataFolder:FindFirstChild("Currency")

    if currency then
        return currency.Value
    end

    return 0
end

local function getAmmoCount()

    local dataFolder = LocalPlayer:FindFirstChild("DataFolder")
    local inventory = dataFolder and dataFolder:FindFirstChild("Inventory")

    if inventory then

        for _, v in ipairs(inventory:GetChildren()) do

            if string.find(v.Name, "Drum%-Shotgun") then
                return tonumber(v.Value) or 0
            end

        end

    end

    return 0
end

local function formatNumber(value)

    local left, num, right =
        string.match(tostring(value), '^([^%d]*)(%d+)(.-)$')

    if not num then
        return tostring(value)
    end

    return left
        .. (
            num
                :reverse()
                :gsub('(%d%d%d)', '%1,')
                :reverse()
                :gsub('^,', '')
        )
        .. right
end

local StartTime = os.time()
local StartCash = getCurrentMoney()
local LastWebhookTime = 0

local http_request_func =
    request
    or http_request
    or (syn and syn.request)
    or (fluxus and fluxus.request)

local function SendWebhookUpdate()

    local currentCash = getCurrentMoney()
    local cashEarned = currentCash - StartCash

    local timeElapsed = os.time() - StartTime
    local hoursElapsed = timeElapsed / 3600

    local cashPerHour =
        hoursElapsed > 0
        and math.floor(cashEarned / hoursElapsed)
        or 0

    if UI.CashRate then
        UI.CashRate.Text =
            "$"
            .. formatNumber(cashPerHour)
            .. " / hr"
    end

    if not getgenv().WEBHOOK_URL
        or getgenv().WEBHOOK_URL == ""
        or not http_request_func then
        return
    end

    if os.time() - LastWebhookTime <
        (getgenv().WEBHOOK_COOLDOWN or 60) then
        return
    end

    LastWebhookTime = os.time()

    local embedData = {
        ["embeds"] = {{
            ["title"] = "Skido Autofarm Progress Update",
            ["color"] = 65280,
            ["fields"] = {
                {
                    ["name"] = "Account",
                    ["value"] = "`" .. LocalPlayer.Name .. "`",
                    ["inline"] = true
                },
                {
                    ["name"] = "Total Wallet Balance",
                    ["value"] = "$" .. formatNumber(currentCash),
                    ["inline"] = true
                },
                {
                    ["name"] = "Cash Earned This Session",
                    ["value"] = "$" .. formatNumber(cashEarned),
                    ["inline"] = false
                },
                {
                    ["name"] = "Estimated Cash / Hour",
                    ["value"] = "$" .. formatNumber(cashPerHour) .. "/hr",
                    ["inline"] = true
                },
                {
                    ["name"] = "Time Active",
                    ["value"] = math.floor(timeElapsed / 60) .. " minutes",
                    ["inline"] = true
                }
            },
            ["footer"] = {
                ["text"] =
                    "Skido AutoFarm System â€¢ "
                    .. os.date("%X")
            }
        }}
    }

    pcall(function()

        http_request_func({
            Url = getgenv().WEBHOOK_URL,
            Method = "POST",
            Headers = {
                ["content-type"] = "application/json"
            },
            Body = HttpService:JSONEncode(embedData)
        })

    end)
end

local function GetCash()

    local Found = {}

    local ignored = workspace:FindFirstChild("Ignored")
    local dropParent = ignored and ignored:FindFirstChild("Drop")

    if not dropParent then
        return Found
    end

    local character = LocalPlayer.Character
    local hrp =
        character
        and character:FindFirstChild("HumanoidRootPart")

    if not hrp then
        return Found
    end

    local children = dropParent:GetChildren()

    for i = 1, #children do

        local v = children[i]

        if v.Name == "MoneyDrop"
            and not pickedUpCash[v]
            and v:IsA("BasePart") then

            if (v.Position - hrp.Position).Magnitude
                <= (getgenv().CASH_AURA_RANGE + 20) then

                table.insert(Found, v)

            end

        end

    end

    return Found
end

        return nil

            return item

local function getPartCFrame(instance)

    if instance:IsA("BasePart") then
        return instance.CFrame
    end

    if instance:IsA("Model") then

        if instance.PrimaryPart then
            return instance.PrimaryPart.CFrame
        end

        local bp = instance:FindFirstChildWhichIsA("BasePart")

        if bp then
            return bp.CFrame
        end

    end

    return nil
end

    end

        getgenv().ammoBoughtCount < 4

            findShopItem("18 %[Drum%-Shotgun Ammo%]")

                    getgenv().ammoBoughtCount < 4

local function executeLootSequence(
    hum,
    hrp,
    currentLootTarget
)

    UpdateUIState("Picking up cash")

    LocalPlayer.CameraMode =
        Enum.CameraMode.Classic

    hum:UnequipTools()

    task.wait(0.3)

    local pickupStart = tick()

    while (tick() - pickupStart) < 20 do

        if isCuffed
            or not getgenv().isLooting
            or (
                currentLootTarget
                and getgenv().selectedCashier
                    ~= currentLootTarget
            ) then

            break
        end

        local CashItems = GetCash()

        if #CashItems == 0 then
            break
        end

        for i = 1, #CashItems do

            if isCuffed
                or not getgenv().isLooting
                or (
                    currentLootTarget
                    and getgenv().selectedCashier
                        ~= currentLootTarget
                ) then

                break
            end

            local v = CashItems[i]

            if v
                and v.Parent
                and not pickedUpCash[v] then

                local cd =
                    v:FindFirstChildOfClass("ClickDetector")

                if cd and hrp then

                    pickedUpCash[v] = true

                    while v
                        and v.Parent
                        and _fireclickdetector
                        and not isCuffed do

                        hrp.CFrame =
                            CFrame.new(
                                v.Position
                                - Vector3.new(0, 3, 0)
                            )

                        hrp.AssemblyLinearVelocity =
                            Vector3.new(0, 0, 0)

                        hrp.AssemblyAngularVelocity =
                            Vector3.new(0, 0, 0)

                        local randomOffset =
                            Vector3.new(
                                math.random(-5, 5),
                                math.random(3, 6),
                                math.random(-5, 5)
                            )

                        workspace.CurrentCamera.CFrame =
                            CFrame.lookAt(
                                v.Position + randomOffset,
                                v.Position
                            )

                        _fireclickdetector(cd, 0)

                        task.wait(0.05)

                        if not v or not v.Parent then
                            break
                        end

                    end

                    task.wait(PICKUP_DELAY)

                end

            end

        end

        RunService.RenderStepped:Wait()

    end

    task.wait(0.3)
end

local function IsCashierAvailable(v)

    if not v then
        return false
    end

    if not v.Name:match("CA%$HIER") then
        return false
    end

    if v.Name:match("VAULT") then
        return false
    end

    local hum = v:FindFirstChildOfClass("Humanoid")
    local head = v:FindFirstChild("Head")

    if not hum or not head then
        return false
    end

    if hum.Health <= 25 then
        return false
    end

    -- Preserve your original excluded location.
    if (
        head.Position
        - Vector3.new(
            -623.998474,
            24.9000168,
            -285.062225
        )
    ).Magnitude <= 8 then
        return false
    end

    return true
end

local function FindAvailableATM()

    local cashiersFolder =
        workspace:FindFirstChild("Cashiers")

    if not cashiersFolder then
        return nil
    end

    for _, v in ipairs(cashiersFolder:GetChildren()) do

        if IsCashierAvailable(v) then
            return v
        end

    end

    return nil
end

-- ====================================================================
-- FIND HIGH WAITING POSITION
-- ====================================================================
-- Finds a high stationary map surface.
--
-- Priority:
-- 1. Large anchored/collidable map parts
-- 2. Building roofs / platforms
-- 3. Terrain surface fallback
--
-- It deliberately ignores the character, Cashiers, dropped money,
-- vehicles and other dynamic objects.
-- ====================================================================

local function IsIgnoredForHighPoint(part)

    if not part or not part:IsA("BasePart") then
        return true
    end

    if not part.CanCollide then
        return true
    end

    if part:IsDescendantOf(LocalPlayer.Character or Instance.new("Folder")) then
        return true
    end

    local parent = part.Parent

    if parent then

        if parent.Name == "Cashiers"
            or parent.Name == "Drop"
            or parent.Name == "Vehicles" then
            return true
        end

    end

    -- Ignore tiny decorative pieces.
    if part.Size.X < 5 and part.Size.Z < 5 then
        return true
    end

    -- Ignore extremely thin/decorative objects.
    if part.Size.Y < 0.2 then
        return true
    end

    return false
end

local function FindHighestMapPart()

    local highestPart = nil
    local highestY = -math.huge

    local descendants =
        workspace:GetDescendants()

    local checked = 0

    for _, obj in ipairs(descendants) do

        if checked >= HIGH_WAIT_SEARCH_LIMIT then
            break
        end

        if obj:IsA("BasePart")
            and not IsIgnoredForHighPoint(obj) then

            checked = checked + 1

            local topY =
                obj.Position.Y
                + (obj.Size.Y / 2)

            -- Avoid absurd objects far above the playable map.
            if topY < 2000
                and topY > highestY then

                highestY = topY
                highestPart = obj

            end

        end

    end

    return highestPart
end

local function FindTerrainWaitingPosition()

    local character = LocalPlayer.Character

    local hrp =
        character
        and character:FindFirstChild("HumanoidRootPart")

    if not hrp then
        return nil
    end

    local origin = hrp.Position

    local rayParams = RaycastParams.new()

    rayParams.FilterType =
        Enum.RaycastFilterType.Exclude

    rayParams.FilterDescendantsInstances = {
        character
    }

    -- Search several nearby terrain directions.
    local offsets = {
        Vector3.new(0, 0, 0),
        Vector3.new(50, 0, 0),
        Vector3.new(-50, 0, 0),
        Vector3.new(0, 0, 50),
        Vector3.new(0, 0, -50),
        Vector3.new(100, 0, 100),
        Vector3.new(-100, 0, -100),
        Vector3.new(100, 0, -100),
        Vector3.new(-100, 0, 100)
    }

    local bestResult = nil

    for _, offset in ipairs(offsets) do

        local startPosition =
            origin
            + offset
            + Vector3.new(0, 1000, 0)

        local result =
            workspace:Raycast(
                startPosition,
                Vector3.new(0, -2000, 0),
                rayParams
            )

        if result
            and result.Instance
            and result.Position.Y > -1000 then

            if not bestResult
                or result.Position.Y
                    > bestResult.Position.Y then

                bestResult = result

            end

        end

    end

    if bestResult then

        return bestResult.Position
            + Vector3.new(
                0,
                HIGH_WAIT_HEIGHT_OFFSET,
                0
            )

    end

    return nil
end

local function GetHighWaitingPosition()

    local highestPart =
        FindHighestMapPart()

    if highestPart then

        local topY =
            highestPart.Position.Y
            + highestPart.Size.Y / 2

        return Vector3.new(
            highestPart.Position.X,
            topY + HIGH_WAIT_HEIGHT_OFFSET,
            highestPart.Position.Z
        )

    end

    return FindTerrainWaitingPosition()
end

-- ====================================================================
-- WAIT FOR ATM
-- ====================================================================

local function MoveToATMWaitingPosition()

    if waitingPosition then
        return
    end

    local position =
        GetHighWaitingPosition()

    if not position then

        local char = LocalPlayer.Character

        local hrp =
            char
            and char:FindFirstChild("HumanoidRootPart")

        if hrp then

            -- Emergency fallback:
            -- stay at current position but raise slightly.
            position =
                hrp.Position
                + Vector3.new(0, 100, 0)

        end

    end

    if position then

        waitingPosition = position

        local char = LocalPlayer.Character

        local hrp =
            char
            and char:FindFirstChild("HumanoidRootPart")

        if hrp then

            hrp.CFrame =
                CFrame.new(waitingPosition)

            hrp.AssemblyLinearVelocity =
                Vector3.new(0, 0, 0)

            hrp.AssemblyAngularVelocity =
                Vector3.new(0, 0, 0)

        end

    end
end

local function WaitForAvailableATM()

    if isWaitingForATM then
        return
    end

    isWaitingForATM = true

    getgenv().selectedCashier = nil
    getgenv().isLooting = false
    getgenv().noclipCashier = false

    LocalPlayer.CameraMode =
        Enum.CameraMode.Classic

    UpdateUIState("NO ATM - WAITING")

    MoveToATMWaitingPosition()

    while isWaitingForATM do

        local char = LocalPlayer.Character

        local hum =
            char
            and char:FindFirstChildOfClass("Humanoid")

        local hrp =
            char
            and char:FindFirstChild("HumanoidRootPart")

        if not char or not hum or not hrp then
            task.wait(NO_ATM_SCAN_DELAY)
            continue
        end

        -- Stop waiting immediately when an ATM appears.
        local availableATM =
            FindAvailableATM()

        if availableATM then

            getgenv().selectedCashier =
                availableATM

            isWaitingForATM = false
            waitingPosition = nil

            UpdateUIState("ATM FOUND")

            task.wait(0.15)

            return availableATM
        end

        -- Make sure the character remains at the waiting location.
        if waitingPosition then

            if (
                hrp.Position
                - waitingPosition
            ).Magnitude > 15 then

                hrp.CFrame =
                    CFrame.new(waitingPosition)

                hrp.AssemblyLinearVelocity =
                    Vector3.new(0, 0, 0)

                hrp.AssemblyAngularVelocity =
                    Vector3.new(0, 0, 0)

            end

        else

            MoveToATMWaitingPosition()

        end

        task.wait(NO_ATM_SCAN_DELAY)

    end

    return nil
end

                        Enum.CameraMode.Classic

                    getgenv().isLooting = false

                        "UpperTorso",

                        "LeftUpperLeg",
                        "RightUpperLeg",
                        "RightUpperArm",
                        "RightLowerArm",
                        "LeftLowerArm",
                        "LeftUpperArm"

                        if part then
                            part:Destroy()
                        end

                    return
                end

                -- ====================================================

                    getCurrentMoney()

                    currentMoney >= 20000

                        "[Drum-Shotgun]"

                        "[Drum-Shotgun]"
                    )

                    hasGun
                    and getAmmoCount() <= 0

                    and canAffordGunfarm then

                        getgenv().isLooting = false
                        getgenv().noclipCashier = false

                            Enum.CameraMode.Classic

                    end

                else

                    if currentMoney < 20000 then
                        getgenv().ammoBoughtCount = 0
                    end

                -- ====================================================
                -- CURRENT CASHIER
                -- ====================================================

                local selectedCashier =
                    getgenv().selectedCashier

                if selectedCashier
                    and (
                        selectedCashier:FindFirstChild("Head")
                        or getgenv().isLooting
                    ) then

                    -- =================================================
                    -- LOOT
                    -- =================================================

                    if getgenv().isLooting
                        or (
                            selectedCashier:FindFirstChild(
                                "Humanoid"
                            )
                            and selectedCashier.Humanoid.Health < 1
                        ) then

                        if not getgenv().isProcessingCash then

                            getgenv().isProcessingCash = true
                            getgenv().isLooting = true

                            local currentLootTarget =
                                selectedCashier

                            executeLootSequence(
                                hum,
                                hrp,
                                currentLootTarget
                            )

                            if getgenv().selectedCashier
                                == currentLootTarget then

                                getgenv().selectedCashier = nil
                                getgenv().noclipCashier = false

                            end

                            getgenv().isProcessingCash = false
                            getgenv().isLooting = false

                            task.wait(1.3)

                        end

                    -- =================================================
                    -- ATTACK
                    -- =================================================

                    elseif selectedCashier:FindFirstChild("Humanoid")
                        and selectedCashier.Humanoid.Health >= 1 then

                        if selectedCashier:FindFirstChild("Open") then

                            local cashierCF =
                                selectedCashier.Open.CFrame

                            local targetPos

                            if getgenv().Gunfarm
                                and canAffordGunfarm
                                and not outOfAmmo then

                                UpdateUIState(
                                    "Attacking with Gun..."
                                )

                                targetPos =
                                    (
                                        cashierCF
                                        * CFrame.new(
                                            0,
                                            0.5,
                                            3.4
                                        )
                                    ).Position

                                hrp.CFrame =
                                    CFrame.lookAt(
                                        targetPos,
                                        cashierCF.Position
                                    )

                            else

                                UpdateUIState(
                                    "Attacking with Combat..."
                                )

                                targetPos =
                                    (
                                        cashierCF
                                        * CFrame.new(
                                            0,
                                            5.5,
                                            0
                                        )
                                    ).Position

                                hrp.CFrame =
                                    CFrame.new(targetPos)
                                    * CFrame.Angles(
                                        math.rad(-90),
                                        0,
                                        0
                                    )

                            end

                            hrp.AssemblyLinearVelocity =
                                Vector3.new(0, 0, 0)

                            hrp.AssemblyAngularVelocity =
                                Vector3.new(0, 0, 0)

                        end

                        if not getgenv().noclipCashier then

                            getgenv().noclipCashier = true

                            for _, v in pairs(
                                selectedCashier:GetDescendants()
                            ) do

                                if v:IsA("BasePart") then
                                    v.CanCollide = false
                                end

                            end

                        end

                        -- =================================================
                        -- CAMERA
                        -- =================================================

                        if getgenv().Gunfarm
                            and canAffordGunfarm
                            and not outOfAmmo then

                            LocalPlayer.CameraMode =
                                Enum.CameraMode.LockFirstPerson

                            if selectedCashier:FindFirstChild("Head") then

                                local targetPosition =
                                    selectedCashier.Head.Position
                                    - Vector3.new(0, 1.6, 0)

                                Camera.CFrame =
                                    CFrame.lookAt(
                                        Camera.CFrame.Position,
                                        targetPosition
                                    )

                            end

                        else

                            LocalPlayer.CameraMode =
                                Enum.CameraMode.Classic

                        end

                        -- =================================================
                        -- WEAPON
                        -- =================================================

                        local weaponName =
                            (
                                getgenv().Gunfarm
                                and canAffordGunfarm
                                and not outOfAmmo
                            )
                            and "[Drum-Shotgun]"
                            or "Combat"

                        local attackTool =
                            LocalPlayer.Backpack:FindFirstChild(
                                weaponName
                            )
                            or char:FindFirstChild(
                                weaponName
                            )

                        if attackTool then

                            if attackTool.Parent
                                == LocalPlayer.Backpack then

                                attackTool.Parent = char

                            end

                            if attackTool:IsA("Tool")
                                and attackTool.Name
                                    ~= "Combat" then

                                local clips =
                                    attackTool:FindFirstChild(
                                        "Clips"
                                    )
                                    or attackTool:FindFirstChild(
                                        "Ammo"
                                    )

                                if clips
                                    and clips.Value == 0 then

                                    game:GetService(
                                        "ReplicatedStorage"
                                    ).MainEvent:FireServer(
                                        "Reload",
                                        attackTool
                                    )

                                    task.wait(0.2)

                                end

                            end

                            local attacking =
                                char:FindFirstChild(
                                    "BodyEffects"
                                )
                                and char.BodyEffects:FindFirstChild(
                                    "Attacking"
                                )

                            if attacking
                                and attacking.Value == false then

                                attackTool:Activate()

                            elseif not char:FindFirstChild(
                                "BodyEffects"
                            ) then

                                attackTool:Activate()

                            end

                        end

                    end

                -- ====================================================
                -- NO CURRENT CASHIER
                -- ====================================================

                elseif not getgenv().isLooting then

                    UpdateUIState("Scanning ATM")

                    LocalPlayer.CameraMode =
                        Enum.CameraMode.Classic

                    -- Search for an ATM/CA$HIER.
                    local foundATM =
                        FindAvailableATM()

                    if foundATM then

                        getgenv().selectedCashier =
                            foundATM

                        isWaitingForATM = false
                        waitingPosition = nil

                        UpdateUIState("ATM FOUND")

                    else

                        -- =============================================
                        -- NO SERVER HOP
                        -- =============================================
                        --
                        -- Instead of:
                        --
                        --     serverHop()
                        --
                        -- the character goes to a high map position
                        -- and waits for a new ATM.
                        --

                        WaitForAvailableATM()

                    end

                end

            end

        end)

    end

end)

-- ====================================================================

-- ====================================================================

local function setupListener(char)

    local be =
        char:WaitForChild("BodyEffects", 5)

    local cuff =
        be
        and be:WaitForChild("Cuff", 5)

    if cuff then

        cuff.Changed:Connect(function(val)

            isCuffed = val

            if isCuffed then

                LocalPlayer.CameraMode =
                    Enum.CameraMode.Classic

                getgenv().selectedCashier = nil
                getgenv().isLooting = false
                getgenv().noclipCashier = false

                isWaitingForATM = false
                waitingPosition = nil

                task.spawn(function()

                    while isCuffed do

                        pcall(BuyAndUseKey)

                        task.wait(0.5)

                    end

                end)

            end

        end)

    end

end
